\section{Yêu cầu của đề tài}

\subsection{Các bên liên quan và câu chuyện người dùng (Stakeholders và User stories)}
Để bảo đảm hệ thống đáp ứng đúng cả góc độ trải nghiệm người chơi lẫn yêu cầu kỹ thuật của một đồ án nghiên cứu AI và Multiplayer, nhóm xác định các bên liên quan (Stakeholders) chính:
\begin{itemize}
    \item \textbf{Người chơi (Player):} Đối tượng trực tiếp xây dựng đội hình, trải nghiệm các chế độ đấu PvE, PvP Offline, PvP Online và tùy biến nhân vật 3D.
    \item \textbf{Khán giả (Spectator):} Người tham gia phòng Online ở vai trò quan sát trực tiếp diễn biến ván đấu của hai người chơi chính mà không can thiệp vào luồng điều khiển.
    \item \textbf{Nhà phát triển và nghiên cứu AI (Developer / AI Researcher):} Nhóm phát triển sử dụng chính lõi trò chơi ở chế độ không đồ họa (Headless) để chạy hàng nghìn ván mô phỏng tự động, huấn luyện mô hình máy học và xuất dữ liệu kiểm chuẩn.
    \item \textbf{Giảng viên hướng dẫn và Hội đồng đánh giá:} Đánh giá tính đúng đắn của kiến trúc, mức độ hoàn thiện của nguyên mẫu giai đoạn 1 và độ tin cậy của các số liệu thực nghiệm.
\end{itemize}

\subsection{Yêu cầu chức năng}
Các yêu cầu chức năng được phân tách thành năm phân hệ chính tương ứng với cấu trúc mã nguồn của dự án:

\subsubsection{Nhóm chức năng thiết lập đội hình và vòng lặp gameplay cốt lõi}
\begin{itemize}
    \item \textbf{FR-01 (Quản lý đội hình):} Cung cấp màn hình Squad Builder cho phép kéo thả hoặc đặt quân cờ vào 12 ô sân nhà (ngoại trừ ô Vua cố định tại chỉ số $3$), tự động kiểm tra ràng buộc ngân sách theo \texttt{GameplayConfig.MaxSlotsInSquad} (hiện cấu hình $12$ Slot Points) và trạng thái mở khóa quân (\texttt{ActivePieces}) thông qua \texttt{SquadValidator}, đồng thời lưu trữ cục bộ xuống đĩa bằng \texttt{SaveLoadManager} (\texttt{SquadSLO}).
    \item \textbf{FR-02 (Khởi tạo bàn cờ và vật cản):} Tự động nạp đội hình hai bên theo \texttt{ISquadProvider}, đặt hai quân Vua cố định và sinh ngẫu nhiên $1$--$2$ ô vật cản tại vùng trung tuyến (hàng 2--3) theo hạt giống \texttt{Seed} xác định.
    \item \textbf{FR-03 (Thẩm định và thực thi nước đi):} Kiểm tra chính xác tính hợp lệ của di chuyển, tấn công, vượt vật cản, hợp thể (khóa \textit{Anti-Same-Type} và \textit{Single-Merge Lock}), phong cấp Tốt, tương tác khiên \texttt{DivineShield}/\texttt{ShieldBreaker}, hồi/trừ AP và trạng thái \texttt{Surge}.
    \item \textbf{FR-04 (Điều kiện kết thúc và bỏ lượt):} Hỗ trợ lệnh bỏ lượt (\texttt{Pass}) có kiểm soát số lần tối đa (\texttt{MaxPassTurn}), tự động kết thúc trận khi Vua bị tiêu diệt, xử thua khi phe tới lượt không còn hành động hợp lệ nào (\texttt{!HasAnyLegalActionForPlayer}) hoặc xử hòa khi đạt \texttt{TurnCount >= MaxTurns}. Phần khung bản tin đầu hàng (\texttt{MSG\_SURRENDER}) đã được định nghĩa ở tầng truyền tải và sẽ nối mạch UI hoàn chỉnh ở Giai đoạn 2.
\end{itemize}

\subsubsection{Nhóm chức năng trí tuệ nhân tạo và kiểm chuẩn (PvE và AI Research)}
\begin{itemize}
    \item \textbf{FR-05 (Hỗ trợ đa tác nhân AI):} Cho phép gắn linh hoạt bất kỳ loại Bot nào vào vị trí người chơi 1 hoặc người chơi 2 thông qua cùng một giao diện \texttt{IMoveInputProvider}: \texttt{RandomBotController}, \texttt{AgentBotController} (vừa chạy suy luận ML-Agents \texttt{.onnx}, vừa chạy Heuristic/Minimax khi bật \texttt{BehaviorType.HeuristicOnly} kết hợp lớp tĩnh \texttt{MinimaxSearch}), \texttt{NeuralMinimaxBotController}, \texttt{NNUEBotController}, \texttt{HybridAgentBotController} và \texttt{LLMBotController}.
    \item \textbf{FR-06 (Hạ tầng huấn luyện và đối kháng tự động):} Cung cấp các Scene Bootstrapper chuyên biệt (\texttt{BehaviourCloneBootstrapper}, \texttt{DaggerBootstrapper}, \texttt{NNUEBootstrapper}, \texttt{BenchmarkBootstrapper}) để thu thập tập dữ liệu chuyên gia, huấn luyện tự đấu đa bàn cờ và ghi nhận tự động kết quả đối kháng (thắng/thua/hòa, thời gian trung bình mỗi nước đi).
\end{itemize}

\subsubsection{Nhóm chức năng nhiều người chơi (PvP Offline và PvP Online)}
\begin{itemize}
    \item \textbf{FR-07 (Chế độ PvP Offline):} Hai người chơi thao tác luân phiên trên cùng thiết bị, hỗ trợ nạp đội hình độc lập cho từng bên và tự động xoay bàn cờ $180^\circ$ sau mỗi lượt bằng \texttt{BoardRotator}.
    \item \textbf{FR-08 (Quản lý phòng Online và phân vai):} Cho phép chọn khu vực máy chủ Photon, nhập mã phòng để tạo hoặc tham gia phiên chơi (\texttt{GameMode.Shared}), đổi vị trí giữa 3 vai trò (Trắng, Đen, Khán giả), đồng bộ tên người chơi, trạng thái sẵn sàng và đội hình hai bên.
    \item \textbf{FR-09 (Đồng bộ trận đấu Online):} Khi trận bắt đầu, máy chủ phòng (Host -- Shared Mode Master Client) phát tán đồng bộ hạt giống vật cản và đội hình (\texttt{MSG\_MATCH\_INIT}); trong trận, mọi lệnh di chuyển và bỏ lượt được truyền tin cậy dưới dạng gói nhị phân cố định 4 bytes (\texttt{MSG\_MOVE}) kèm bộ đếm thời gian lượt đi \texttt{OnlineTurnTimer} (60 giây/lượt; luồng tự động xử lý khi hết giờ \texttt{OnTimerExpired} và nút đầu hàng sẽ nối mạch hoàn chỉnh ở Giai đoạn 2).
\end{itemize}

\subsubsection{Nhóm chức năng tầng hiển thị, giao diện và âm thanh}
\begin{itemize}
    \item \textbf{FR-10 (Diễn hoạt tách rời logic):} Toàn bộ sự kiện thay đổi trạng thái ván đấu được gom thành \texttt{GameEventBatch} và phát lần lượt qua hàng đợi trực quan \texttt{VisualQueueManager} để kích hoạt chuyển động, hoạt họa 3D, hiệu ứng rung/zoom camera, VFX và âm thanh mà không làm đơ luồng logic.
    \item \textbf{FR-11 (HUD và cài đặt):} Hiển thị trực quan lượng AP hiện có, lượng AP dự kiến hồi lượt sau, số lượt Pass còn lại, lượt đi hiện tại, nghĩa địa quân cờ bị tiêu diệt (Graveyard) và hộp thoại cài đặt \texttt{UIPopupSettings} (Giai đoạn 1 đã hoàn thiện tab General lưu tên người chơi qua \texttt{GeneralSettingsSLO}; các tab Sound/Graphics/Input đã dựng khung \texttt{UIPopupSettingsTab} chờ mở rộng ở Giai đoạn 2).
\end{itemize}

\subsubsection{Nhóm chức năng mở rộng tùy biến nhân vật 3D (Smart Customization)}
\begin{itemize}
    \item \textbf{FR-12 (Tái tạo khuôn mặt và tóc 3D từ ảnh 2D):} Cho phép người dùng chọn ảnh chân dung 2D và giới tính (Nam/Nữ) trên \texttt{SCCanvasUI}; \texttt{SCCharacterCreator} khởi chạy tiến trình suy luận cục bộ \texttt{infer\_onnx.py} (CLI subprocess qua \texttt{ProcessStartInfo}) và nạp tệp kết quả \texttt{sc\_result.json}, ảnh da khử bóng cùng bản đồ pháp tuyến (Normal Map) để áp 25 trọng số hình thái khuôn mặt trực giao (BlendShapes), màu mống mắt và màu tóc tự động; đồng thời hỗ trợ tinh chỉnh thủ công bằng thanh trượt và chuyển đổi giữa 7 lựa chọn kiểu tóc 3D (6 mẫu tóc MPFB v2 và không tóc).
\end{itemize}

\subsection{Yêu cầu phi chức năng}
\begin{itemize}
    \item \textbf{Tính tách biệt và khả năng bảo trì (Decoupling và Maintainability):} Các lớp thực thể và thẩm định luật cốt lõi (\texttt{GameState}, \texttt{Piece}, \texttt{MoveValidator}, \texttt{PassiveFlags}, \texttt{SquadValidator}) phải tách khỏi vòng đời \texttt{MonoBehaviour} và đồ họa 3D, chỉ đọc tham số tập trung qua \texttt{GameConfig}, bảo đảm cùng một trạng thái $S_t$ và lệnh $C_t$ luôn sinh ra trạng thái $S_{t+1}$ giống hệt nhau (Deterministic) trên mọi máy khách và môi trường huấn luyện.
    \item \textbf{Hiệu năng tính toán (Performance):}
    \begin{itemize}
        \item Chế độ chơi thực tế duy trì khung hình ổn định ($\ge 60$ FPS trên cấu hình PC tầm trung GPU RTX 4050 Laptop).
        \item Thời gian phản hồi trung bình của Bot AI trong trận đấu thực tế dưới $0.5$ giây/nước đi (mục tiêu dưới $100$ ms với Minimax/NeuralMinimax/Residual NNUE ở độ sâu 4).
        \item Chế độ Headless training tắt toàn bộ render cho phép mô phỏng hàng trăm trận đấu trong vài chục giây mà không phát sinh rò rỉ bộ nhớ.
    \end{itemize}
    \item \textbf{Hiệu quả băng thông mạng (Network Efficiency):} Tối ưu kích thước gói tin đồng bộ nước đi xuống mức tối thiểu (dưới $100$ bytes cho chuỗi 10 lần thăm dò) để bảo đảm tính ổn định kể cả trên đường truyền mạng thông thường.
    \item \textbf{Độ trễ xử lý Smart Customization:} Thời gian suy luận toàn trình trên GPU cục bộ (trích mốc MediaPipe + hồi quy ONNX B1 v2 + BVLS + chiếu Barycentric PyTorch3D + Bilateral Symmetry Delighting) cần hoàn tất trong vài giây (đo thực tế khoảng $5.9$ giây toàn trình cho ảnh chân dung đầu vào) để người dùng xem trước trực tiếp trong game.
\end{itemize}

\subsection{Yêu cầu kỹ thuật và trải nghiệm người dùng}
\begin{itemize}
    \item \textbf{Nền tảng mục tiêu:} Hệ điều hành Windows (PC Standalone 64-bit), sử dụng Unity 6 (URP -- Universal Render Pipeline) \cite{unity6}, ngôn ngữ C\# (.NET Standard 2.1) kết hợp thư viện bất đồng bộ UniTask và môi trường Python 3.10+ (PyTorch, PyTorch3D \cite{pytorch3d}, ONNX Runtime, MediaPipe \cite{mediapipe}, OpenCV/SciPy) cho huấn luyện AI và pipeline suy luận cục bộ Smart Customization.
    \item \textbf{Trải nghiệm người dùng (UX):} Phản hồi thị giác và thính giác tức thời khi di chuột qua ô cờ, chọn quân, di chuyển, va chạm khiên hoặc hợp thể; bố cục UI thích ứng theo độ phân giải màn hình (\texttt{CanvasScaler}) và hiển thị rõ ràng trạng thái kinh tế AP để giảm tải nhận thức cho người mới tiếp cận luật hợp thể.
\end{itemize}
